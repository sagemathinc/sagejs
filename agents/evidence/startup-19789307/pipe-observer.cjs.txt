   if(process.platform==="win32")spawnSync("taskkill.exe",["/pid",String(child.pid),"/t","/f"],{stdio:"ignore",timeout:3000});

  }
 });

  if(done)return;done=true;clearTimeout(timer);
  if(code!==0||signal||phase!==2||errors.trim())return reject(Error("failed answer/exit: "+errors.slice(0,400)));

})}
